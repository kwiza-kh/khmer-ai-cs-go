## ការរៀបចំការដ្ឋាន និងការផ្ទុកផ្ទាំង

មុនពេលដំឡើង ត្រូវរៀបចំការដ្ឋានឱ្យបានត្រឹមត្រូវ៖

- ត្រូវមានផ្ទៃរាបស្មើ និងស្អាតសម្រាប់ដាក់ផ្ទាំង ដោយដាក់គ្នាយ ឈើ ឬដែកចន្លោះប្រហែល 1 m និងលើកឱ្យខ្ពស់ពីដីយ៉ាងតិច 100 mm។
- ត្រូវរៀបចំផ្លូវដើរសម្រាប់ការលើកផ្ទាំងពីឡានទៅទីតាំងដំឡើង ដោយគ្មានឧបសគ្គ និងគ្មានជំរាលខ្ពស់។
- ត្រូវពិនិត្យអាកាសធាតុ។ នៅកម្ពុជា រដូវវស្សាមានភ្លៀងធ្លាក់ភ្លាមៗ ដូច្នេះការដំឡើងដំបូល និងការផ្សាភ្ជាប់សន្លាក់គួរធ្វើក្នុងរយៈពេលដែលអាចបញ្ចប់មុនភ្លៀង។
- ផ្ទាំងត្រូវរក្សាទុកក្នុងម្លប់ លើផ្ទៃខ្ពស់ពីដី ឆ្ងាយពីអណ្ដាតភ្លើង និងសារធាតុរំលាយ។ ការប៉ះពន្លឺព្រះអាទិត្យដោយផ្ទាល់ ភ្លៀងយូរ ឬសារធាតុរំលាយ និងប្រេង មិនស្ថិតក្នុងការធានាទេ។
- ត្រូវពិនិត្យប័ណ្ណបញ្ជូនទំនិញឱ្យបានច្បាស់ ដែលបញ្ជាក់ដង់ស៊ីតេ វិមាត្រ បរិមាណ និងលេខក្រុមផលិត។ បើឃើញផ្ទាំងខូច ឬខ្វះ ត្រូវលើកឡើងភ្លាមៗ និងមិនឱ្យលើសពី **3 ថ្ងៃធ្វើការ** បន្ទាប់ពីការដឹកជញ្ជូន ជាមួយរូបថត ប័ណ្ណបញ្ជូន និងលេខក្រុមផលិត។

## ការត្រួតពិនិត្យផ្ទៃទ្រនាប់ និងស៊ុមគ្រោង

ផ្ទាំងសាំងវិចជាប់នឹងស៊ុមគ្រោង (ដែករង ដែកឈរ ឬជញ្ជាំងខាងក្រោម) ដូច្នេះគុណភាពនៃស៊ុមកំណត់គុណភាពនៃផ្ទាំង៖

- ត្រូវពិនិត្យថាផ្ទៃទ្រនាប់មានកម្រិតស្មើគ្នា។ គម្លាតលើសពី 5 mm ក្នុងរង្វាស់ 2 m ត្រូវកែដោយបន្ថែមស្លាបព្រា ឬកែស៊ុមជាមុន។
- ត្រូវពិនិត្យភាពឈរត្រង់ នៃដែកឈរ និងភាពជ្រុងកែង 90° នៅជ្រុងបន្ទប់។ បើស៊ុមរអិល ផ្ទាំងនឹងបើកចន្លោះនៅសន្លាក់ ហើយមិនអាចបិទជិតបានទេ។
- ត្រូវពិនិត្យជម្រាយបង្ហូរទឹកនៅលើដំបូល។ សម្រាប់ផ្ទាំងដំបូល ត្រូវមានជម្រាលយ៉ាងតិច 5 % ទៅរកទិសបង្ហូរទឹក ហើយមិនត្រូវមានផ្ទៃចុះទាបដែលធ្វើឱ្យទឹកជំាងនៅតែមួយកន្លែង។
- ត្រូវពិនិត្យថាគ្រឿងបរិក្ខារភ្ជាប់ មានគុណភាពល្អ និងមិនមានចំណុចច្រែះដែលអាចចម្លងទៅផ្ទាំង។
- ត្រូវបញ្ជាក់ថាគ្មានការងារភ្លើង ឬការងារផ្សារនៅជិតទីតាំងដាក់ផ្ទាំង។ បើចាំបាច់ត្រូវធ្វើការងារក្ដៅ ត្រូវបំពេញមុនពេលយកផ្ទាំងចូល និងមានឧបករណ៍ពន្លត់ភ្លើងនៅនឹងកន្លែង។

## ការលើក និងការកាន់ផ្ទាំងដោយមិនធ្វើឱ្យខូចគែម

គែមផ្ទាំងសាំងវិចមានដែកស្ដើង និងស្នូលពពុះនៅខាងក្នុង ដូច្នេះការប៉ះទង្គិច ឬការទាញខូចគែមគឺជាមូលហេតុទូទៅបំផុតនៃការខូចផ្ទាំងនៅការដ្ឋាន៖

- ប្រើឧបករណ៍លើកដោយខ្យល់ ឬឃ្នាស់កាន់គែម សម្រាប់ផ្ទាំងធំ និងជាពិសេសផ្ទាំងបន្ទប់ត្រជាក់កម្រាស់ 150–200 mm។
- មិនត្រូវលើកផ្ទាំងដោយចងខ្សែតែមួយចំហៀង ឬដោយទាញពីគែមតែមួយ ព្រោះផ្ទាំងនឹងពត់ និងបែកសន្លាក់ស្អិត។
- ត្រូវមនុស្សពីរនាក់ឡើងទៅសម្រាប់ការលើកផ្ទាំងវែង ហើយលើកឱ្យត្រង់ មិនឱ្យផ្ទាំងរមៀល ឬកោង។
- មិនត្រូវអូសផ្ទាំងលើដី ឬលើគែមផ្ទាំងផ្សេងទេ។ ត្រូវលើកជានិច្ច។
- ត្រូវដាក់បន្ទះទន់ (ក្រដាសកាតុង ឬស្នូលពពុះ) នៅចំណុចដែលខ្សែ ឬឃ្នាស់ប៉ះគែមផ្ទាំង។
- នៅពេលដាក់ផ្ទាំងចុះ ត្រូវដាក់ជាន់លើគ្នាឱ្យត្រង់ ហើយមិនត្រូវដាក់របស់ធ្ងន់លើផ្ទាំង ដែលអាចធ្វើឱ្យស្នូលរួញ។
- កុំឱ្យផ្ទាំងសើម ឬប៉ះទឹកយូរ។ ស្នូល EPS ស្រូបទឹកបាន ≤ 2 % តាមមាឌ ប៉ុន្តែទឹកដែលជ្រាបចូលសន្លាក់នឹងបំផ្លាញសមត្ថភាពអ៊ីសូឡង់ និងបង្កច្រែះលើស្រទាប់ដែក។

## វិធីភ្ជាប់ និងគំលាតខ្ចៅ

ការភ្ជាប់ផ្ទាំងទៅស៊ុមគ្រោងត្រូវធ្វើឡើងដោយខ្ចៅដែកខ្លួនឯង ដែលមានប្រដាប់ប្រក់កៅស៊ូ នៅក្រោមក្បាលខ្ចៅ ដើម្បីការពារទឹកជ្រាបចូលតាមរន្ធខ្ចៅ៖

- ខ្ចៅត្រូវចាប់នៅលើកំពូលរលក មិនមែននៅបាតរលកទេ ព្រោះការចាប់នៅបាតរលកធ្វើឱ្យទឹកជំាង និងជ្រាបចូល។
- គំលាតខ្ចៅតាមបណ្ដោយគែមផ្ទាំង៖ ប្រហែល 500 mm។ នៅតំបន់ដែលមានខ្យល់បក់ខ្លាំង ឬផ្ទាំងខ្ពស់ គួរបង្កើនដង់ស៊ីតេខ្ចៅ ដោយប្រើគំលាត 300–400 mm នៅជិតគែម និងជ្រុង។
- ចម្ងាយខ្ចៅពីគែមផ្ទាំង៖ យ៉ាងតិច 25 mm ដើម្បីកុំឱ្យដែកគែមរហែក។
- ប្រវែងខ្ចៅ៖ ត្រូវលើសកម្រាស់ផ្ទាំងយ៉ាងតិច 25–30 mm ដើម្បីចាប់ចូលស៊ុមដែកបានជ្រៅគ្រប់គ្រាន់។
- ការបង្វិលខ្ចៅ៖ ត្រូវបង្វិលឱ្យកៅស៊ូ ចុចសង្កត់ជាប់នឹងផ្ទាំង ប៉ុន្តែមិនត្រូវបង្វិលលើស ដែលធ្វើឱ្យកៅស៊ូខូច ឬដែកផ្ទាំងរួញចូល។
- ការពង្រីកដោយកម្ដៅ៖ ផ្ទាំងដែករីក និងរួញតាមកម្ដៅ។ ត្រូវទុកចន្លោះតូច 2–3 mm នៅចុងផ្ទាំង និងមិនត្រូវចាប់ខ្ចៅជាប់ជាដាច់ខាតទាំងសងខាងដោយគ្មានចន្លោះ។ សូមចងចាំថា ស្ថេរភាពវិមាត្ររបស់ EPS គឺ ≤ 0.5 % នៅ 70 °C រយៈពេល 48 ម៉ោង។
- នៅក្នុងបន្ទប់ត្រជាក់ ត្រូវប្រើខ្ចៅ ឬគ្រឿងភ្ជាប់ដែលកាត់ផ្ដាច់ការចម្លងកម្ដៅ ដើម្បីកុំឱ្យកម្ដៅចម្លងតាមខ្ចៅចូលបន្ទប់ និងបង្កើតជាចំណុចសាយទឹក។

## ការផ្សាភ្ជាប់សន្លាក់ប្រឆាំងសំណើម និងខ្យល់ចេញចូល

ការផ្សាភ្ជាប់ជាជំហានដែលកំណត់អាយុកាលនៃសំណង់។ នៅអាកាសធាតុកម្ពុជា ដែលមានសំណើមខ្ពស់ និងសីតុណ្ហភាពខ្ពស់ ខ្យល់សើមដែលចូលទៅដល់ស្នូល EPS នឹងបង្កើតជាទឹកខាប់នៅខាងក្នុង ហើយបំផ្លាញសមត្ថភាពអ៊ីសូឡង់បន្តិចម្ដងៗ៖

- ត្រូវដាក់សារធាតុផ្សាភ្ជាប់ពីរជាន់៖ ជាន់ទីមួយនៅបាតចង្អូរ (ខាងក្នុង) និងជាន់ទីពីរនៅខាងក្រៅ។ ការដាក់តែជាន់ក្រៅ មិនការពារខ្យល់ចូលតាមចន្លោះខាងក្នុងបានទេ។
- ប្រើសារធាតុផ្សាភ្ជាប់ប្រភេទស៊ីលីកូនបន្សាបអាស៊ីត ឬបន្ទះកាវប៊ូទីល។ **មិនត្រូវប្រើកាវ ឬសារធាតុផ្សាភ្ជាប់ដែលមានសារធាតុរំលាយ ប៉ះនឹង EPS ជាដាច់ខាត** ព្រោះវានឹងរលាយស្នូលពពុះ។
- ត្រូវដាក់បន្ទះបិទជិត នៅចុងផ្ទាំងដំបូល និងនៅចន្លោះរលក ដើម្បីបិទចន្លោះប្រហោងដែលមើលមិនឃើញ។
- សន្លាក់ជ្រុង និងសន្លាក់ជាមួយជញ្ជាំងផ្សេង ត្រូវប្រើបន្ទះគែម និងផ្សាភ្ជាប់ពីរជាន់ដូចគ្នា។
- រាល់រន្ធខ្ចៅ គម្លាតជុំវិញទ្វារ បង្អួច និងបំពង់ឆ្លងកាត់ ត្រូវផ្សាភ្ជាប់ដោយសារធាតុផ្សាភ្ជាប់ ឬបន្ទះកៅស៊ូ។
- បន្ទាប់ពីផ្សាភ្ជាប់រួច ត្រូវពិនិត្យដោយភ្នែក និងដោយដៃ ដើម្បីរកចន្លោះដែលសារធាតុផ្សាភ្ជាប់មិនពេញ ជាពិសេសនៅជ្រុង និងចំណុចប្រសព្វ។
- នៅបន្ទប់ត្រជាក់ គួរធ្វើការតេស្តសំណើម ឬការតេស្តសម្ពាធ មុនបញ្ជូន ដើម្បីបញ្ជាក់ថាសំណង់ជិតល្អ។

## ការកាត់ផ្ទាំងនៅការដ្ឋានដោយសុវត្ថិភាព

ការកាត់នៅការដ្ឋានត្រូវធ្វើដោយឧបករណ៍សមស្រប និងដោយការប្រុងប្រយ័ត្ន ព្រោះកំហុសកាត់ដោយអតិថិជនមិនស្ថិតក្នុងការធានាទេ៖

- កាត់ស្នូល EPS ដោយរណាធ្មេញល្អិត (18–24 ធ្មេញក្នុង 1 អ៊ីញ) ឬដោយខ្សែក្តៅ។ ការកាត់ដោយខ្សែក្តៅផ្ដល់ផ្ទៃរាបស្មើ និងគ្មានធូលី។
- កាត់ស្រទាប់ដែកដោយកន្ត្រៃកាត់ដែក ឬកន្ត្រៃកាត់ប្រវែង។ មិនត្រូវប្រើម៉ាស៊ីនកាត់ដែកដោយថាស ទេ ព្រោះផ្កាភ្លើងក្ដៅអាចដុតថ្នាំស្រោប និងបង្កគ្រោះថ្នាក់ភ្លើងជាមួយ EPS។
- ត្រូវវាស់ និងគូសគំនូសសង្ខេបច្បាស់លាស់ មុនកាត់ ហើយកាត់សាកល្បងមួយសន្លឹកមុន ដើម្បីផ្ទៀងផ្ទាត់ទំហំ។
- ត្រូវយកស្នូល EPS ចេញប្រហែល 50 mm នៅចុងផ្ទាំង ដើម្បីបង្កើតជាសន្លាក់ ឬដាក់បន្ទះបិទ តាមការណែនាំរបស់ក្រុមបច្ចេកទេស។
- ត្រូវសម្អាតសំណល់ដែក ចេញពីផ្ទាំងភ្លាមៗ ព្រោះសំណល់ដែកនឹងបង្កើតជាចំណុចច្រែះលើផ្ទៃដែកស្រោបពណ៌ ហើយជាប់នៅលើផ្ទាំងយូរ។
- ត្រូវខាត់គែមដែកដែលមុត ដើម្បីការពារការរបួសដៃ និងដើម្បីកុំឱ្យគែមមុតកាត់បន្ទះផ្សាភ្ជាប់។
- ត្រូវពាក់ម៉ាស់ វ៉ែនតា មដាន និងសម្លៀកបំពាក់វែងដៃ ពេលកាត់ និងខួង។
- ត្រូវកាត់នៅកន្លែងដែលមានខ្យល់ចេញចូលល្អ និងឆ្ងាយពីប្រភពភ្លើង។
- ត្រូវចងចាំថា ផ្ទាំងដែលកាត់រួច គួរដំឡើងឱ្យបានលឿន និងមិនត្រូវទុកចោលឱ្យប៉ះភ្លៀង ឬពន្លឺព្រះអាទិត្យយូរ។

## លំដាប់ការងារ៖ ជញ្ជាំងមុន ពិដានក្រោយ

លំដាប់ការងារត្រឹមត្រូវជួយកាត់បន្ថយការខូចខាត និងការងារធ្វើឡើងវិញ៖

1. **ការវាស់ និងការកំណត់ទីតាំង**: គូសបញ្ជាក់ទីតាំងជញ្ជាំង ជ្រុង ទ្វារ បង្អួច និងបំពង់ឆ្លងកាត់លើកម្រាល។
2. **ការដំឡើងប្រអប់កម្រាល**: ដំឡើងជើងទម្រ និងប្រអប់ខាងក្រោមឱ្យត្រង់ និងរាបស្មើ ដោយផ្សាភ្ជាប់ជាមួយកម្រាលដើម្បីការពារទឹកជ្រាបពីក្រោម។
3. **ការដំឡើងផ្ទាំងជ្រុង**: ចាប់ផ្ដើមពីជ្រុងមួយ ដោយតម្រូវឱ្យឈរត្រង់ និងជ្រុងកែង 90°។ ជ្រុងជាចំណុចយោងនៃជញ្ជាំងទាំងមូល។
4. **ការដំឡើងផ្ទាំងជញ្ជាំងតាមលំដាប់**: ដំឡើងម្ដងមួយផ្ទាំង ដោយចាក់សោសន្លាក់ តម្រូវភាពឈរត្រង់ជាបន្តបន្ទាប់ និងដាក់ខ្ចៅតាមគំលាតកំណត់។
5. **ការដំឡើងធ្នឹមទ្រពិដាន**: ដំឡើងធ្នឹម ឬស៊ុមទ្រពិដានឱ្យបានរឹង និងរាបស្មើ មុនដំឡើងផ្ទាំងពិដាន។
6. **ការដំឡើងផ្ទាំងពិដាន**: ធ្វើពីវេទិកា ឬរន្ធត់ដែលមានសុវត្ថិភាព ដោយចាប់ផ្ដើមពីចុងម្ខាង និងរំកិលទៅមុខ។ ត្រូវចាក់សោសន្លាក់ឱ្យពេញ និងផ្សាភ្ជាប់ជាបន្តបន្ទាប់។
7. **ការដំឡើងទ្វារ បង្អួច និងឧបករណ៍ឆ្លងកាត់**: កាត់ និងដំឡើងឱ្យត្រូវតាមទំហំ ដោយផ្សាភ្ជាប់ជុំវិញឱ្យជិត។
8. **ការផ្សាភ្ជាប់ជាន់ទីពីរ និងការសម្អាត**: ផ្សាភ្ជាប់សន្លាក់ទាំងអស់ជាន់ទីពីរ សម្អាតសំណល់ដែក និងកាកសំណល់ចេញពីផ្ទាំង។
